# 详细设计

> 游戏核心流程、数据流、状态的实现细节

## 1. 游戏启动流程

```
game.js
  └→ new Main()
       ├→ render.js 初始化 Canvas
       ├→ new DataBus() 全局状态
       ├→ new Background() 背景
       ├→ new Cultivator() 修炼者
       ├→ new CombatSystem() 战斗
       ├→ new CultivationSystem() 修炼
       ├→ new UIManager() UI 管理
       ├→ loadSaveData() 读档
       ├→ calculateOfflineRewards() 离线结算
       └→ this.loop() 启动主循环
```

## 2. 游戏主循环

```js
class Main {
  constructor() {
    this.bg = new Background();
    this.cultivator = new Cultivator();
    this.combat = new CombatSystem();
    this.cultivationSys = new CultivationSystem();
    this.shopSys = new ShopSystem();
    this.ui = new UIManager();

    this.start();
  }

  start() {
    GameGlobal.databus.reset();
    this.aniId = requestAnimationFrame(this.loop.bind(this));
  }

  update(deltaTime) {
    GameGlobal.databus.frame++;

    if (GameGlobal.databus.isGameOver) return;

    // 执行顺序很重要：
    // 1. 修炼 tick（每秒执行一次实际逻辑）
    this.cultivationSys.update(deltaTime);

    // 2. 术法释放
    this.cultivator.updateSkills();

    // 3. 战斗系统（怪物生成、攻击判定、掉落）
    this.combat.update(deltaTime);

    // 4. 商店刷新计时
    this.shopSys.update(deltaTime);

    // 5. 每分钟收益更新（每秒 60 帧，60 秒 = 3600 帧）
    if (GameGlobal.databus.frame % 3600 === 0) {
      this.ui.updatePerMinRates();
    }

    // 6. UI 刷新
    this.ui.update();

    // 7. 自动存档（每 600 帧 ≈ 10 秒）
    if (GameGlobal.databus.frame % 600 === 0) {
      saveToStorage();
    }
  }

  render(ctx) {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    this.bg.render(ctx);
    this.combat.render(ctx);     // 怪物 + 伤害数字
    this.cultivator.render(ctx); // 修炼者（在怪物上方）
    this.ui.render(ctx);         // UI 在最上层
  }

  loop() {
    const now = Date.now();
    const deltaTime = (now - (this.lastTime || now)) / 1000;
    this.lastTime = now;

    this.update(deltaTime);
    this.render(canvasCtx);
    this.aniId = requestAnimationFrame(this.loop.bind(this));
  }
}
```

## 4. 修炼系统 Ticker 设计

修炼每秒执行一次（每秒消耗灵石 → 获得修为）：

```js
class CultivationSystem {
  timer = 0;
  TICK_INTERVAL = 1; // 秒

  update(deltaTime) {
    this.timer += deltaTime;
    if (this.timer < this.TICK_INTERVAL) return;
    this.timer = 0;

    const player = GameGlobal.databus.cultivator;
    if (!player.autoCultivate) return;

    // 修为已满且小等级=10 → 等待手动突破
    if (player.cultivation >= player.maxCultivation) {
      if (player.level < 10) {
        this.levelUp(player); // 自动升小等级
      }
      return; // level=10 时等待突破
    }

    // 计算灵石消耗
    const stoneCost = player.cultivationSpeed * player.stoneCostPerSec;
    if (player.spiritStone < stoneCost) {
      GameGlobal.databus.cultivationPaused = true;
      return;
    }

    // 消耗灵石，增加修为
    player.spiritStone -= stoneCost;
    player.cultivation += player.cultivationSpeed;

    // 灵力自动回复
    if (player.spirit < player.maxSpirit) {
      player.spirit = Math.min(player.maxSpirit, player.spirit + player.spiritRegen);
    }

    // 修炼中积累心法熟练度
    player.innerSkills.forEach(skill => {
      skill.proficiency += 1;
      checkProficiencyLevelUp(skill);
    });
  }

  levelUp(player) {
    player.level += 1;
    player.cultivation = 0;
    player.maxCultivation = this.calcCultivationRequired(player.realmIndex, player.level);
    this.updatePlayerStats(player); // 按小等级更新属性
  }
}
```

## 5. 战斗系统设计

```js
class CombatSystem {
  constructor() {
    this.monsterSpawnTimer = 0;
    this.monsterRespawnCooldown = 2; // 死亡后 2 秒重生
    this.currentMonster = null;
  }

  update(deltaTime) {
    this.updateMonsterSpawn();
    this.updateCombat();
    this.updateDamageNumbers();
  }

  updateMonsterSpawn() {
    // 没有怪物时生成新怪物
    if (!this.currentMonster || !this.currentMonster.isActive) {
      this.monsterSpawnTimer += deltaTime;
      if (this.monsterSpawnTimer >= this.monsterRespawnCooldown) {
        this.monsterSpawnTimer = 0;
        this.spawnMonster();
      }
    }
  }

  spawnMonster() {
    const monster = GameGlobal.databus.pool.getItemByClass('monster', Monster);
    monster.init(GameGlobal.databus.cultivator.realmIndex);
    GameGlobal.databus.monsters.push(monster);
    this.currentMonster = monster;
  }

  updateCombat() {
    if (!this.currentMonster || !this.currentMonster.isActive) return;

    const player = GameGlobal.databus.cultivator;

    // 普通攻击计时
    player.attackTimer += deltaTime;
    if (player.attackTimer >= player.attackSpeed) {
      player.attackTimer = 0;
      this.performNormalAttack(player, this.currentMonster);
    }

    // 主动术法：灵力足够就释放，无 CD 限制
    player.outerSkills
      .filter(s => s.type === 'active')
      .forEach(skill => {
        if (player.spirit >= skill.spiritCost) {
          player.spirit -= skill.spiritCost;
          this.performSkillAttack(player, this.currentMonster, skill);
        }
      });
  }

  performNormalAttack(player, monster) {
    const damage = this.calcDamage(player, monster, 1);
    monster.hp -= damage;
    this.spawnDamageNumber(monster.x, monster.y, damage, false);
    player.state = 'attack';
    this.checkMonsterDeath(monster);
  }

  performSkillAttack(player, monster, skill) {
    const damage = this.calcDamage(player, monster, skill.getEffect());
    monster.hp -= damage;
    this.spawnDamageNumber(monster.x, monster.y, damage, false, 'skill');
    skill.proficiency += 1; // 术法熟练度
    checkProficiencyLevelUp(skill);
    this.checkMonsterDeath(monster);
  }

  calcDamage(attacker, defender, multiplier) {
    const base = Math.floor(attacker.attack ** 2 / (attacker.attack + defender.defense));
    let damage = Math.floor(base * multiplier);

    // 暴击
    if (Math.random() < attacker.critRate) {
      damage *= attacker.critDamage;
      // 标记为暴击，DamageNumber 渲染时用黄色
    }
    return Math.floor(damage);
  }

  checkMonsterDeath(monster) {
    if (monster.hp <= 0) {
      monster.isActive = false;
      monster.state = 'dead';

      // 发放奖励
      const player = GameGlobal.databus.cultivator;
      player.spiritStone += monster.stoneReward;
      player.spirit = Math.min(player.maxSpirit, player.spirit + 5); // 击杀回复少量灵力

      // 掉落装备和残卷（直接进背包）
      this.rollEquipmentDrop(monster);
      this.rollFragmentDrop(monster);

      // 回收怪物
      GameGlobal.databus.pool.recover('monster', monster);
      GameGlobal.databus.totalKills++;
      this.currentMonster = null;
    }
  }
}
```

## 6. 商店系统设计

```js
class ShopSystem {
  REFRESH_INTERVAL = 4 * 3600; // 4 小时（现实时间）
  timer = 0;
  items = [];        // 当前商品列表

  update(deltaTime) {
    this.timer += deltaTime;
    if (this.timer >= this.REFRESH_INTERVAL) {
      this.autoRefresh();
    }
  }

  autoRefresh() {
    this.timer = 0;
    const player = GameGlobal.databus.cultivator;
    const slotCount = player.getGlobalLevel(); // 人物等级 = 商品数

    // 保留锁定的商品
    const locked = this.items.filter(item => item.locked);
    const freeSlots = slotCount - locked.length;

    // 在空闲槽位生成新商品
    const newItems = [];
    for (let i = 0; i < freeSlots; i++) {
      newItems.push(this.generateItem(player.realmIndex));
    }

    this.items = [...locked, ...newItems];
  }

  manualRefresh() {
    const cost = this.getRefreshCost();
    const player = GameGlobal.databus.cultivator;
    if (player.spiritStone < cost) return;
    player.spiritStone -= cost;
    this.autoRefresh();
  }

  getRefreshCost() {
    return 50 * (GameGlobal.databus.cultivator.realmIndex + 1);
  }

  toggleLock(index) {
    this.items[index].locked = !this.items[index].locked;
  }

  buy(index) {
    const item = this.items[index];
    const player = GameGlobal.databus.cultivator;
    if (player.spiritStone < item.price) return false;

    player.spiritStone -= item.price;
    // 物品进背包 / 功法习得
    if (item.type === 'equipment') {
      addToBag(item.data);
    } else if (item.type === 'skill_fragment') {
      addFragment(item.data);
    } else if (item.type === 'skill_complete') {
      learnSkill(item.data);
    }

    this.items.splice(index, 1); // 购买后移除
    return true;
  }

  generateItem(realmIndex) {
    // 按权重随机生成：装备/术法残卷/心法残卷/完整功法
    const roll = Math.random() * 100;
    if (roll < 50) return this.genEquipment(realmIndex);
    if (roll < 75) return this.genSkillFragment('outer');
    if (roll < 90) return this.genSkillFragment('inner');
    return this.genCompleteSkill();
  }
}
```

## 7. 存档与读档

```js
const SAVE_KEY = 'cultivation_idle_save';

function saveToStorage() {
  const player = GameGlobal.databus.cultivator;
  const data = {
    realmIndex: player.realmIndex,
    cultivation: player.cultivation,
    maxCultivation: player.maxCultivation,
    spiritStone: player.spiritStone,
    equipment: player.equipment,
    innerSkills: player.innerSkills.map(s => ({
      id: s.id, proficiency: s.proficiency, proficiencyLevel: s.proficiencyLevel,
    })),
    outerSkills: player.outerSkills.map(s => ({
      id: s.id, proficiency: s.proficiency, proficiencyLevel: s.proficiencyLevel,
    })),
    lastOnlineTime: Date.now(),
    totalKills: GameGlobal.databus.totalKills,
  };
  wx.setStorageSync(SAVE_KEY, data);
}

function loadSaveData() {
  const data = wx.getStorageSync(SAVE_KEY);
  if (!data) return null; // 新玩家

  // 恢复修炼者状态
  const player = GameGlobal.databus.cultivator;
  player.realmIndex = data.realmIndex;
  player.cultivation = data.cultivation;
  player.spiritStone = data.spiritStone;
  // ... 恢复装备、心法、术法

  // 计算离线收益
  if (data.lastOnlineTime) {
    const offlineSec = (Date.now() - data.lastOnlineTime) / 1000;
    calculateOfflineRewards(offlineSec);
  }

  return data;
}
```

## 8. 离线收益计算

```js
function calculateOfflineRewards(offlineSec) {
  // 无离线系数，满额收益
  const offlineMin = Math.floor(offlineSec / 60);
  const player = GameGlobal.databus.cultivator;

  // 每分钟修炼收益（与在线一致）
  const cultivationPerMin = player.cultivationSpeed * 60;
  const cultivationGain = cultivationPerMin * offlineMin;

  // 每分钟战斗灵石收益
  const combatStonePerMin = player.cultivationSpeed * player.stoneCostPerSec * 3 * 60;
  const stoneGain = combatStonePerMin * offlineMin;

  // 扣除修炼灵石消耗
  const cultivationCost = player.cultivationSpeed * player.stoneCostPerSec * offlineSec;
  const netStone = Math.max(0, stoneGain - cultivationCost);

  return {
    totalMin: offlineMin,
    cultivation: cultivationGain,
    stone: netStone,
    stoneSpent: cultivationCost,
  };
}
```
